Apply a policy, e.g.:
  kubectl apply -f - <<'EOF'
  apiVersion: ebfw.dvrkn.com/v1
  kind: EgressPolicy
  metadata: { name: allowlist, namespace: default }
  spec:
    defaultAction: Deny
    rules:
      - name: allow-dns
        action: Allow
        match: { ports: [53] }
      - name: allow-github
        action: Allow
        match: { domains: ["github.com", "*.githubusercontent.com"], ports: [443] }
  EOF

WARNING: a ClusterEgressPolicy with defaultAction: Deny default-denies the whole
node — be sure to allow the API server, image registries, and DNS first.
